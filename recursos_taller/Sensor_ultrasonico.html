hcsr04_distance.ino
// Medición acústica Time-of-Flight
const int pinTrig = 3;
const int pinEcho = 2;
long duracionUs;
float distanciaCm;

void setup() {
  pinMode(pinTrig, OUTPUT);
  pinMode(pinEcho, INPUT);
  Serial.begin(9600);
}

void loop() {
  // 1. Disparo de 10µs en TRIG
  digitalWrite(pinTrig, LOW);
  delayMicroseconds(2);
  digitalWrite(pinTrig, HIGH);
  delayMicroseconds(10);
  digitalWrite(pinTrig, LOW);

  // 2. Medir duración del pulso con pulseIn()
  duracionUs = pulseIn(pinEcho, HIGH, 38000);
  distanciaCm = (duracionUs * 0.0343) / 2.0;

  if (distanciaCm >= 2.0 && distanciaCm <= 400.0) {
    Serial.print("Distancia: ");
    Serial.print(distanciaCm);
    Serial.println(" cm");
  }
  else {
    Serial.println("Fuera de rango (Sin eco)");
  }
  delay(60);
}
